 #include  <WiFi.h> 
 #include  <HTTPClient.h> 
 #include  <WiFiClientSecure.h> 
 #include  <ArduinoJson.h>

const char* WIFI_SSID = WIFI_SSID_VALUE;
const char* WIFI_PASS = WIFI_PASSWORD_VALUE;

const char* API_URL = "https://httpbin.org/anything";

void connectWiFi()
{
  Serial.println();
  Serial.print("Connecting to Wi-Fi: ");
  Serial.println(WIFI_SSID);

  WiFi.mode(WIFI_STA);
  WiFi.begin(WIFI_SSID, WIFI_PASS);

  int retryCount = 0;

  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");

    retryCount++;

    if (retryCount > 40) {
      Serial.println();
      Serial.println("Wi-Fi connection failed.");
      return;
    }
  }

  Serial.println();
  Serial.println("Wi-Fi connected.");
  Serial.print("IP address: ");
  Serial.println(WiFi.localIP());
}

void parseHttpbinResponseHttpBin(const String& payload)
{
  JsonDocument doc;
  DeserializationError error = deserializeJson(doc, payload);

  if (error) {
    Serial.print("Response JSON parse failed: ");
    Serial.println(error.c_str());
    return;
  }

  const char* url = doc["url"];

  Serial.println("---- Response Summary ----");
  Serial.print("url: ");
  Serial.println(url);
}

void httpGetFromHttpBinAsJson()
{
  if (WiFi.status() != WL_CONNECTED) {
    Serial.println("Wi-Fi is not connected.");
    return;
  }

  //String jsonData = createPostJson();

  Serial.println();
  Serial.println("Starting HTTP GET request...");
  Serial.print("URL: ");
  Serial.println(API_URL);
  Serial.println();

  WiFiClientSecure client;
  client.setInsecure();

  HTTPClient http;
  bool beginResult = http.begin(client, API_URL);

  if (!beginResult) {
    Serial.println("HTTPClient begin failed.");
    return;
  }

  http.setTimeout(10000);
  http.addHeader("Content-Type", "application/json");

  int httpCode = http.GET();

  Serial.print("HTTP status code: ");
  Serial.println(httpCode);

  if (httpCode == HTTP_CODE_OK) {
    String payload = http.getString();

    Serial.println("---- Response Body ----");
    Serial.println(payload);

    parseHttpbinResponseHttpBin(payload);
  } else {
    Serial.print("HTTP POST failed. Error: ");
    Serial.println(http.errorToString(httpCode));
  }

  http.end();
}

void setup()
{
  Serial.begin(115200);
  Serial.println("setup() begin");

  connectWiFi();

  httpGetFromHttpBinAsJson();

  Serial.println("func: setup() done");
}

void loop()
{

}
